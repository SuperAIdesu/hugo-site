{{- /* Renders one date or date range for a section entry. Context: dict page page dateFormat Go date layout (default "Jan 2006") showRange append the end date, or "Present" when endDate is empty */ -}} {{- $page := .page -}} {{- $format := .dateFormat | default "Jan 2006" -}} {{- $showRange := .showRange | default false -}} {{- if not $page.Date.IsZero -}} {{ $page.Date.Format $format }} {{- if $showRange }} – {{- with $page.Params.endDate }} {{- $end := time.AsTime . }} {{ $end.Format $format }} {{- else }} Present {{- end }} {{- end }} {{- end -}}
